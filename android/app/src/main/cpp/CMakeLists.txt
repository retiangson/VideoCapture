cmake_minimum_required(VERSION 3.22.1)
project(videocapture_speech)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_FLAGS_DEBUG "-O3 -g")
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g")
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../../../vendor/whisper.cpp-d1be6fde11ac6e0407606b4e42fe72d34add8037 whisper)
add_library(videocapture_speech SHARED speech.cpp)
target_link_libraries(videocapture_speech whisper android log)
target_compile_options(videocapture_speech PRIVATE -O3)
target_link_options(videocapture_speech PRIVATE -Wl,-z,max-page-size=16384)
